const express = require("express") const jwt = require("jsonwebtoken") const bcrypt = require("bcryptjs") const bodyParser = require("body-parser") const app = express() app.use(bodyParser.json()) /* ========================= 🧠 FAKE DATABASE (replace later w/ Firebase / JSONBin / Keyv) ========================= */ const db = { users: [], announcements: [], auditLogs: [] } /* ========================= 🔐 CONFIG ========================= */ const JWT_SECRET = "super_secret_change_this_lol" /* ========================= 🧑‍💻 ROLES ========================= */ const ROLES = { STUDENT: "student", PARENT: "parent", TEACHER: "teacher", ADMIN: "admin", DISTRICT_ADMIN: "district_admin" } /* ========================= 🧠 AUDIT LOG SYSTEM ========================= */ function audit(action, data) { db.auditLogs.push({ action, data, time: new Date().toISOString() }) } /* ========================= 🔒 AUTH MIDDLEWARE ========================= */ function auth(req, res, next) { const token = req.headers["authorization"] if (!token) return res.status(401).send("no token bro") try { const decoded = jwt.verify(token.split(" ")[1], JWT_SECRET) req.user = decoded next() } catch (e) { return res.status(403).send("invalid token 💀") } } /* ========================= 🧠 ROLE GUARD ========================= */ function requireRole(roles) { return (req, res, next) => { if (!roles.includes(req.user.role)) { return res.status(403).send("no perms lil bro") } next() } } /* ========================= 👤 REGISTER ========================= */ app.post("/register", async (req, res) => { const { username, password, role } = req.body const hash = await bcrypt.hash(password, 10) const user = { id: db.users.length + 1, username, password: hash, role: role || ROLES.STUDENT } db.users.push(user) audit("USER_CREATED", { username, role }) res.send("user made 👍") }) /* ========================= 🔑 LOGIN (JWT) ========================= */ app.post("/login", async (req, res) => { const { username, password } = req.body const user = db.users.find(u => u.username === username) if (!user) return res.status(400).send("no user") const match = await bcrypt.compare(password, user.password) if (!match) return res.status(400).send("wrong password") const token = jwt.sign( { id: user.id, role: user.role }, JWT_SECRET, { expiresIn: "2h" } ) res.json({ token }) }) /* ========================= 📢 ANNOUNCEMENTS (ADMIN ONLY) ========================= */ app.post( "/announce", auth, requireRole([ROLES.ADMIN, ROLES.DISTRICT_ADMIN]), (req, res) => { const { message } = req.body const announcement = { id: db.announcements.length + 1, message, by: req.user.role, time: new Date().toISOString() } db.announcements.push(announcement) audit("ANNOUNCEMENT_CREATED", announcement) res.send("announcement dropped 📢") } ) /* ========================= 📜 GET ANNOUNCEMENTS ========================= */ app.get("/announcements", auth, (req, res) => { res.json(db.announcements) }) /* ========================= 📊 AUDIT LOGS (ADMIN ONLY) ========================= */ app.get( "/logs", auth, requireRole([ROLES.ADMIN, ROLES.DISTRICT_ADMIN]), (req, res) => { res.json(db.auditLogs) } ) /* ========================= 👥 USERS LIST (ADMIN ONLY) ========================= */ app.get( "/users", auth, requireRole([ROLES.ADMIN, ROLES.DISTRICT_ADMIN]), (req, res) => { res.json(db.users.map(u => ({ ...u, password: "hidden 💀" }))) } ) /* ========================= 🧨 BASIC FRONTEND DASHBOARD ========================= */ app.get("/", (req, res) => { res.send(`

📚 School Portal

`) }) /* ========================= 🚀 START SERVER ========================= */ app.listen(3000, () => { console.log("school system running on http://localhost:3000") })
